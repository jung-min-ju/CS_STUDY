Observer pattern
interface Observer {
update(data): void
}
class Subject {
private observers: Observer[] = []
subscribe(obs) {
this.observers.push(obs)
}
unsubscribe(obs) {
this.observers = this.observers.filter(o => o !== obs)
}
notify(data) {
this.observers.forEach(o => o.update(data))
}
}
