이 화면에서는Model·View·Controller 세 박스가 역할을, 화살표(User action → Controller → Model update → View render)가 데이터 흐름을 보여줍니다. "+ 할 일 추가" 버튼으로 Todo 시뮬레이션을 실행해 보세요.
언제 떠올리나같은 데이터를 여러 화면에서 보여주거나, 디자이너와 백엔드 개발자가 각자 맡은 영역을 독립적으로 작업해야 할 때.
MVC pattern
// Model — 데이터와 비즈니스 로직
class TodoModel {
private todos = []
addTodo(text) { this.todos.push({ text, done: false }) }
toggle(i) { this.todos[i].done = !this.todos[i].done }
getTodos() { return [...this.todos] }
}
// View — 화면 표시
class TodoView {
render(todos) { /* DOM 업데이트 */ }
bindAddBtn(handler) { /* 클릭 이벤트 연결 */ }
}
// Controller — Model과 View를 연결
class TodoController {
constructor(model, view) {
view.bindAddBtn(text => {
model.addTodo(text)
view.render(model.getTodos())
})
}
}
