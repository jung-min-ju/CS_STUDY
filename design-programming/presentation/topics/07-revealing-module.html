무슨 패턴인가IIFE 클로저 안에 변수와 함수를 숨기고, return으로 공개할 것만 골라 내보냅니다. 외부에서 내부 상태를 직접 건드리지 못하게 막아요.
이 화면에서는private 영역(count, MAX, validate)과 public 영역(increment, getCount)이 나뉘어 있습니다. 초록색 public 버튼은 정상 호출되고, 빨간색 private 버튼(module.count, module.validate() 등)을 누르면 Access denied 카운터가 올라갑니다.
언제 떠올리나ES Module을 쓰기 어려운 레거시 환경에서 전역 네임스페이스 오염 없이 캡슐화가 필요할 때. 유틸리티 라이브러리나 jQuery 플러그인 구조에서 자주 쓰입니다.
Revealing module pattern
const counterModule = (function() {
// private 영역 — 외부 접근 불가
let count = 0
const MAX = 10
function validate(n) {
return n >= 0 && n <= MAX
}
function increment() {
if (validate(count + 1)) count++
}
function getCount() {
return count
}
// public API — 이것만 노출!
return { increment, getCount }
})()
